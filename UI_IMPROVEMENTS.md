```javascript
  email: '',
```

### Form Handlers
```javascript
const handleFormChange = (e) => {
  setFormData({
    ...formData,
    [e.target.name]: e.target.value
  });
};

const handleFormSubmit = (e) => {
  e.preventDefault();
  console.log('Form submitted:', formData);
  setSnackbar({ 
    open: true, 
    message: 'Thank you for your interest!' 
  });

};
```
